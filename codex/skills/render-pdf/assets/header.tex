% hyperref as raw color NAMES (not hex) — unlike titlepage-color, which the
% template wraps in \definecolor itself. Define a named color here so
% frontmatter can reference "linkaccent" instead of a raw hex string.

% Pandoc 3.9 sets \LTcaptype to "none" to suppress table numbering,
% but the caption package tries to use "none" as a counter, which fails.
% Newer TeX Live already defines \c@none in some configurations, so guard it.
\ifcsname c@none\endcsname\else\newcounter{none}\fi

  }
\fi
